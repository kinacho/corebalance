		expect(1).toBe(1);
	});

	it('afirma que un literal existe', () => {
		expect(true).toBeTruthy();
	});

	it('no afirma absolutamente nada', () => {
		const resultado = 2 + 2;
		console.log(resultado);
	});

	// Éstos son legítimos y NO deben salir en el informe.

	});
