# CoreBalance es una herramienta gratuita y el contenido del blog es material de
# difusión: nos interesa que los buscadores clásicos y los generativos lo lean,
# lo citen y lo recomienden. Por eso el acceso es abierto salvo dos excepciones:
#   1. Las rutas privadas o sin valor de indexación (/api/, /dashboard/).
#   2. Los scrapers que revenden datos sin enviar tráfico ni alimentar modelos
#      que puedan recomendarnos (ver bloque final).

# ── 1. Regla general: buscadores, asistentes y bots de entrenamiento ──────────
# Incluye Googlebot, Bingbot (base de ChatGPT Search y Copilot), GPTBot,
# ClaudeBot, OAI-SearchBot, PerplexityBot, Google-Extended (AI Overviews y
# AI Mode), Applebot-Extended, meta-externalagent, Amazonbot, CCBot, etc.
# Sin barra final a propósito: `Disallow: /dashboard/` NO cubre `/dashboard`, que es
# la URL real. Bingbot la rastreó por esa rendija y la marcó como página sin meta
# description — y lo está, porque el dashboard va con `ssr = false` y lo único que
# recibe un rastreador es el esqueleto de `app.html`.

Disallow: /dashboard
